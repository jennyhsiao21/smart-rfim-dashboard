/* ---------- TYPEWRITER ---------- */

function useTypewriter(text, speed=14, run=true){

  const [out, setOut] = useState('');

  const [done, setDone] = useState(false);

  useEffect(()=>{

    if(!run){ setOut(''); setDone(false); return; }

    setOut(''); setDone(false);

    let i = 0;

    const id = setInterval(()=>{

      i += Math.max(1, Math.round(text.length/180)); // chunked for long text

      if(i >= text.length){ setOut(text); setDone(true); clearInterval(id); }

      else setOut(text.slice(0, i));

    }, speed);

    return ()=>clearInterval(id);

  }, [text, speed, run]);

  return [out, done];

}


/* ---------- SPARKLINE ---------- */

function Sparkline({ data, stroke='#0F1115', fill='rgba(15,17,21,0.06)', w=220, h=56 }){

  const min = Math.min(...data), max = Math.max(...data);

  const span = max - min || 1;

  const step = w / (data.length - 1);

  const points = data.map((v,i)=>[i*step, h - ((v-min)/span)*(h-8) - 4]);

  const d = points.map((p,i)=> (i?'L':'M') + p[0].toFixed(1) + ' ' + p[1].toFixed(1)).join(' ');

  const area = `${d} L ${w} ${h} L 0 ${h} Z`;

  const last = points[points.length-1];

  return (

    <svg width={w} height={h} viewBox={`0 0 ${w} ${h}`} className="block">

      <path d={area} fill={fill} />

      <path d={d} className="sparkpath" fill="none" stroke={stroke} strokeWidth="1.6" strokeLinecap="round" strokeLinejoin="round" />

      <circle cx={last[0]} cy={last[1]} r="2.6" fill={stroke} />

    </svg>

  );

}

/* ---------- AI CAMPAIGN WORKFLOW ---------- */

function AIWorkflow({ segment, onReset }){

  // phases: idle | thinking | draft | editing | success

  const [phase, setPhase] = useState('thinking');

  const [thoughtIdx, setThoughtIdx] = useState(0);

  const [edited, setEdited] = useState(null);


  const thoughts = [

    'Analysing segment behaviour patterns…',

    'Cross-referencing repurchase rhythm against cohort benchmarks…',

    'Matching to top-performing campaigns for similar merchants…',

    'Drafting message and calibrating offer depth…',

  ];


  useEffect(()=>{

    setPhase('thinking');

    setThoughtIdx(0);

    setEdited(null);

    const t1 = setTimeout(()=>setThoughtIdx(1), 450);

    const t2 = setTimeout(()=>setThoughtIdx(2), 900);

    const t3 = setTimeout(()=>setThoughtIdx(3), 1250);

    const t4 = setTimeout(()=>setPhase('draft'), 1600);

    return ()=>{ [t1,t2,t3,t4].forEach(clearTimeout); };

  }, [segment.id]);


  const c = segment.campaign;

  const reach = segment.count;

  const expected = Math.round(reach * c.cvr / 100);


  const fullSubject = (edited?.subject ?? c.subject);

  const fullBody    = (edited?.body    ?? c.body);


  const [subjOut, subjDone] = useTypewriter(fullSubject, 12, phase==='draft');

  const [bodyOut, bodyDone] = useTypewriter(fullBody, 6, phase==='draft' && subjDone);


  if (phase === 'thinking') {

    return (

      <div className="rounded-xl bg-ink-900 text-canvas-50 p-5 fade-in">

        <div className="flex items-center gap-3 mb-4">

          <div className="relative w-7 h-7">

            <div className="absolute inset-0 rounded-full border border-canvas-50/15"></div>

          </div>

          {thoughts.map((t, i) => (

            <li key={i} className={`flex items-start gap-3 text-[13.5px] transition-opacity ${i <= thoughtIdx ? 'opacity-100' : 'opacity-25'}`}>

              <span className={`pulse-dot mt-2 ${i < thoughtIdx ? 'bg-emerald-400' : 'bg-canvas-50'}`}

                    style={{ animation: i === thoughtIdx ? undefined : 'none', opacity: i < thoughtIdx ? 1 : undefined }}></span>

              <span className={i < thoughtIdx ? 'text-canvas-50/70 line-through decoration-canvas-50/20' : ''}>

                {t}

              </span>

            </li>

          ))}

        </ul>

      </div>

    );

  }


  if (phase === 'success') {

    const tomorrow = '10:00 AM tomorrow';

    return (

      <div className="rounded-xl bg-white shadow-card hairline p-6 fade-in">

        <div className="flex items-start gap-3">

          <div className="w-8 h-8 rounded-full flex items-center justify-center" style={{background:'#E3F3EA'}}>

            <svg width="16" height="16" viewBox="0 0 24 24" fill="none"><path d="M5 12.5l4 4 10-10" stroke="#1F8A5B" strokeWidth="2.2" strokeLinecap="round" strokeLinejoin="round"/></svg>

          </div>

          <div className="flex-1">

            <div className="text-[15px] font-semibold">Campaign scheduled</div>

            <div className="text-[13.5px] text-ink-500 mt-1">

              <span className="num text-ink-900 font-medium">{fmt(reach)}</span> customers in <span className="font-medium text-ink-900">{segment.name}</span> will receive this message at <span className="font-medium text-ink-900">{tomorrow}</span>.

            <div className="mt-4 flex items-center gap-3">

        <div className="flex items-center gap-3">

          <span className="text-[11px] uppercase tracking-wider text-ink-400" style={{letterSpacing:'.14em'}}>AI Draft · v1</span>

          <span className="inline-flex items-center gap-1.5 text-[11px] px-2 py-0.5 rounded-full bg-canvas-100 text-ink-700">

        </div>

        <div className="text-[11px] text-ink-400 num">{segment.name}</div>

        <div className="text-[10.5px] uppercase tracking-wider text-ink-400" style={{letterSpacing:'.12em'}}>Offer</div>

      <div className="px-5 pt-4 pb-5">

        <div className="text-[10.5px] uppercase tracking-wider text-ink-400" style={{letterSpacing:'.12em'}}>Subject</div>

        {phase === 'editing' ? (

          <input

            className="w-full mt-1 text-[14.5px] font-medium text-ink-900 bg-canvas-50 rounded-md px-3 py-2 hairline focus:outline-none focus:ring-2 focus:ring-accent/40"

          </div>

          <span className="w-1.5 h-1.5 rounded-full bg-emerald-500"></span>

/* ---------- DEEP DIVE PANEL ---------- */

function DeepDive({ segment, onGenerate, generated }){

  const trend = segment.trend;

  const delta = ((trend[trend.length-1] - trend[0]) / trend[0]) * 100;

  const direction = delta >= 0 ? 'up' : 'down';

  const sparkColor = direction === 'up' ? (segment.tone === 'lost' ? '#C8392E' : '#1F8A5B') : '#C8392E';

  // Lost growing is bad; sleeping growing is bad. Color logic: for "risk"/"lost" tones, growth is red.

  const goodGrowth = !(segment.tone === 'risk' || segment.tone === 'lost');

  const finalColor = (delta >= 0)

    ? (goodGrowth ? '#1F8A5B' : '#C8392E')

    : (goodGrowth ? '#C8392E' : '#1F8A5B');


  return (

    <div className="space-y-5">

      {/* header card */}

      <div className="rounded-xl bg-white shadow-card hairline p-5">

        <div className="flex items-start justify-between gap-4">

          <div>

            <div className="flex items-center gap-2">

              <ToneChip tone={segment.tone} />

              <ActionChip kind={segment.actionType} />

            </div>

            <h3 className="text-[20px] font-semibold mt-2 leading-tight" data-comment-anchor="dd-title">{segment.name}</h3>

            <div className="text-[13px] text-ink-500 mt-1">{segment.rule}</div>

          </div>

          <div className="text-right">

            <div className="num text-[28px] font-semibold leading-none">{fmt(segment.count)}</div>

            <div className="text-[12px] text-ink-400 mt-1 num">{pct(segment.count/TOTAL_BASE*100)} of base</div>

          </div>

        </div>


        {/* metrics row */}

        <div className="grid grid-cols-3 mt-5 gap-3">

          <div className="rounded-lg bg-canvas-50 p-3 hairline">

            <div className="text-[10.5px] uppercase tracking-wider text-ink-400" style={{letterSpacing:'.12em'}}>Recommended</div>

            <div className="text-[13.5px] font-medium mt-1 text-ink-900">{segment.action}</div>

            <div className="text-[10.5px] uppercase tracking-wider text-ink-400" style={{letterSpacing:'.12em'}}>7-mo trend</div>

            <div className="text-[10.5px] uppercase tracking-wider text-ink-400" style={{letterSpacing:'.12em'}}>Avg. LTV</div>

            <div className="text-[13.5px] font-medium mt-1 num">${fmt(Math.round(segment.count > 1500 ? 142 : segment.tone==='top' ? 1480 : 320))}</div>

          </div>

        </div>

        <div className="mt-5">

          <div className="flex items-center justify-between mb-1.5">

            <div className="text-[10.5px] uppercase tracking-wider text-ink-400" style={{letterSpacing:'.12em'}}>Customers in segment · last 7 months</div>

          </div>

          <button

            onClick={onGenerate}

            className="text-[13px] px-4 py-2 rounded-md text-canvas-50 inline-flex items-center gap-2 hover:opacity-90"

            style={{background:'#0F1115'}}>

            <svg width="13" height="13" viewBox="0 0 24 24" fill="none"><path d="M12 3l1.7 5.3L19 10l-5.3 1.7L12 17l-1.7-5.3L5 10l5.3-1.7L12 3z" fill="currentColor"/></svg>
